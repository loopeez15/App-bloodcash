-- 1. Crear la tabla de Checklist (sirve para Tareas, Localizaciones y Equipo)
create table if not exists checklist_items (
  id uuid default gen_random_uuid() primary key,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null,
  label text not null,
  category text,
  completed boolean default false,
  list_type text not null -- Valores esperados: 'tasks', 'locations', 'equipment'
);

-- 2. Crear la tabla de Guiones
create table if not exists scripts (
  id uuid default gen_random_uuid() primary key,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null,
  updated_at timestamp with time zone default timezone('utc'::text, now()) not null,
  tape_id text unique not null, -- ID de la cinta (ej: "1", "2")
  content text
);

-- 3. Crear la tabla de Production Bible (Documentación)
create table if not exists production_bible (
  section_key text primary key, -- ej: 'breakdown', 'schedule'
  content text,
  updated_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 4. Habilitar seguridad a nivel de fila (RLS)
alter table checklist_items enable row level security;
alter table scripts enable row level security;
alter table production_bible enable row level security;

-- 5. Crear políticas de acceso PÚBLICO
-- Checklist
create policy "Public Select Checklist" on checklist_items for select using (true);
create policy "Public Insert Checklist" on checklist_items for insert with check (true);
create policy "Public Update Checklist" on checklist_items for update using (true);
create policy "Public Delete Checklist" on checklist_items for delete using (true);

-- Scripts
create policy "Public Select Scripts" on scripts for select using (true);
create policy "Public Insert Scripts" on scripts for insert with check (true);
create policy "Public Update Scripts" on scripts for update using (true);

-- Production Bible
create policy "Public Select Bible" on production_bible for select using (true);
create policy "Public Insert Bible" on production_bible for insert with check (true);
create policy "Public Update Bible" on production_bible for update using (true);

-- --- FIN DEL CÓDIGO SQL ---
